two variables as input:
	- wireLength
		distance from other pixel
	- groupSize
		every other group changes direction of sorting

package main

import (
	"fmt"
)

func main() {
	wireLength, groupSize := 2, 4
	for i := 0; i < 16; i++ {
		getOtherIndex(i, wireLength, groupSize)
	}
	
}

func getOtherIndex(index, wireLength, groupSize int) {
	isBlueGroup := (((index - (index % groupSize)) / groupSize) % 2) == 0
	isFirstPeer := (index % (wireLength * 2)) < wireLength
	fmt.Printf("%v:\t%v\t%v\n", index, isBlueGroup, isFirstPeer)
}
